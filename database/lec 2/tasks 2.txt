-- Create Employees Table
CREATE TABLE Employees (
    EmployeeID INT,
    FirstName VARCHAR(50),
    LastName VARCHAR(50),
    Department VARCHAR(50),
    Salary INT
);

----------------------------------------------------------------

----------------------------------------------------------------

-- Update salary of EmployeeID 101 to 600000
UPDATE Employees
SET Salary = 600000
WHERE EmployeeID = 101;

----------------------------------------------------------------

-- Delete employee where EmployeeID = 101
DELETE FROM Employees
WHERE EmployeeID = 101;

----------------------------------------------------------------

-- Retrieve all employees in IT department
SELECT *
FROM Employees
WHERE Department = 'IT';

----------------------------------------------------------------

-- Select all data with concatenation of FirstName and LastName
SELECT EmployeeID,
       CONCAT(FirstName, ' ', LastName) AS FullName,
       Department,
       Salary
FROM Employees;
